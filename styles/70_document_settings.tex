% Sichtbare Druckseiten-Kennzeichnung für den 2-auf-1-Druck:
% PDF-Seiten 1/2 -> 1a/1b, 3/4 -> 2a/2b, ...
% \thepage bleibt unverändert für PDF-Viewer, Hyperlinks und Druckdialog.
\newcommand{\ZSFPrintPageLabel}{%
  \ifodd\value{page}%
    \number\numexpr(\value{page}+1)/2\relax a%
  \else
    \number\numexpr\value{page}/2\relax b%
  \fi
}

% Build-Kennung unten mittig, Druckseiten-Kennzeichnung unten rechts (overlay,
% layout-neutral). Lokal automatisch DEV + Datum; im Release der SemVer-Tag.
\AddToShipoutPictureFG{%
  \AtPageLowerLeft{%
    \put(\dimexpr\paperwidth/2\relax,1mm){%
      \makebox(0,0)[b]{{\transparent{0.5}\sffamily\bfseries\fontsize{9.5pt}{9.5pt}\selectfont\color{black}\ZSFReleaseLabel}}%
    }%
    \put(\dimexpr\paperwidth-2mm\relax,1mm){%
      \makebox(0,0)[rb]{\sffamily\bfseries\fontsize{9.5pt}{9.5pt}\selectfont\color{black}\ZSFPrintPageLabel}%
    }%
  }%
}
